pipeline {
    agent any

    stages {

        stage('Checkout') {
            steps {
                echo 'Checking out source code...'
            }
        }

        stage('Validate') {
            steps {
                sh '''
                    test -f index.html
                    test -f style.css
                    test -f script.js
                    echo "Website files validated successfully"
                '''
            }
        }

        stage('Deploy') {
            steps {
                sh '''
                    sudo cp index.html /usr/share/nginx/html/
                    sudo cp style.css /usr/share/nginx/html/
                    sudo cp script.js /usr/share/nginx/html/
                    echo "Website deployed successfully"
                '''
            }
        }
    }

    post {
        success {
            echo 'CI/CD Pipeline completed successfully!'
        }

        failure {
            echo 'Pipeline failed!'
        }
    }
}
